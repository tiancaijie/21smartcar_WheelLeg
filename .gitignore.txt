# IAR 项目通用忽略规则
# 1. 调试/编译输出目录（不同项目目录名可能为Debug/Release/Out等，按需调整）
Debug/
Release/
Out/
Build/
obj/
bin/

# 2. IAR 专属工程文件/调试产物
*.ewd        # IAR 工作区调试信息文件
*.ewt        # IAR 工作区配置文件（本地调试配置，无需提交）
*.pbi        # IAR 项目构建信息文件
*.pdb        # IAR 调试符号文件
*.dep        # IAR 依赖关系文件
*.d         # 编译依赖文件
*.o         # 目标文件（编译产物）
*.obj       # 目标文件（Windows平台编译产物）
*.lib       # 静态库文件（本地编译生成，无需提交）
*.a         # 静态库文件（Linux/Mac平台编译产物）
*.lst       # IAR 列表文件（编译日志）
*.map       # 链接映射文件
*.hex       # 烧录文件（本地编译生成，无需提交）
*.bin       # 二进制烧录文件
*.bak       # IAR 备份文件
*.tmp       # 临时文件
*.log       # 本地日志文件
*.sra       # IAR 堆栈分析文件
*.svd       # 本地寄存器配置文件（若为项目通用可取消注释）

# 3. VS Code 本地配置（无需提交到远程仓库）
.vscode/     # VS Code 调试/配置目录（若团队共用配置可单独保留launch.json/tasks.json）
*.code-workspace # VS Code 工作区文件

# 4. 系统临时文件
.DS_Store    # Mac 系统隐藏文件
Thumbs.db    # Windows 系统缩略图文件
